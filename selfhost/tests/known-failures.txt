# Tests where scc and safec are known to disagree. difftest.py still runs

# listed here fails. Reasons are in SUBSET.md, "Not supported" and
# "Differences". A line '<test> wasm' lists it for scc.wasm (--scc-wasm)
# only.

# SIMD vector types are not supported
ctests/simd/vector_align.sc
# malloc_defer (parser sugar) is not supported (SUBSET.md, "Not supported")
ctests/regions/malloc_defer_typed.sc
# safec's import of the host's C headers through clang (<complex.h> here):
# scc has no C header import, only the shims in selfhost/include
ctests/cheaders/complex_float128.sc
# element references into a container's buffer ('v.at(i)', 'm.get(k)')
# are not tracked, so one held across a push or a rehashing insert is not
# reported (SUBSET.md, "Memory-flow analysis")
ctests/lifetimes/container_refs_err.sc

# ownership (Drop) errors: scc doesn't run safec's ownership checks
# (SUBSET.md, "Ownership and thread safety")
ctests/ownership/borrows_err.sc
ctests/ownership/destructors_err.sc
ctests/ownership/forget_err.sc
ctests/ownership/generics_err.sc
ctests/ownership/leaks_err.sc
ctests/ownership/use_after_err.sc
ctests/ownership/use_after_col_err.sc
# std/collections/bst.h is not bundled with scc (SUBSET.md, "Not supported")
ctests/ownership/tree_inserts_time.sc
# std/thread.h is not bundled with scc (SUBSET.md, "Not supported")
ctests/threads/mutex.sc
# scc.wasm (the playground path) compiles one source: a test #including a
